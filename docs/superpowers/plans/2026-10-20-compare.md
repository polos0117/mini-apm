# 무엇이 다른가 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** X-View에서 고른 요청과 나머지를 속성 값별 비율로 비교해 차이가 큰 값부터 보여 준다.

```js
// "무엇이 다른가": 고른 트랜잭션과 나머지를 속성 값별 비율로 비교한다. HTTP와 무관한 순수 로직.
const MAX_ROWS = 15;
const MIN_DIFF = 0.1 - 1e-9; // 10%p 미만은 뺀다(정확히 10%p가 부동소수 오차로 빠지지 않게)
const MAX_VALUE = 200;
const MAX_EXCEPTION = 80;
const NONE = '(없음)';

const cut = (v) => String(v).slice(0, MAX_VALUE);
const or = (v) => (v === null || v === undefined || v === '' ? NONE : cut(v));

// API별 통계와 같은 묶음 키
const routeKey = (tx) => (typeof tx.route === 'string' && tx.route ? tx.route : String(tx.name));

function pathOf(referer) {
  try {
    return cut(new URL(referer).pathname); // 호스트·? 뒤는 뺀다
  } catch {
    return NONE;
  }
}

function hostOf(text) {
  try {
    return cut(new URL(String(text).split(' ')[1]).host); // "GET http://host:port/..."
  } catch {
    return '(알 수 없음)';
  }
}

// 트랜잭션 하나의 속성 값들. 키 "속성\t값"으로 같은 값은 한 번만
function attrs(tx) {
  const req = tx.req && typeof tx.req === 'object' ? tx.req : {};
  const list = [
    ['앱', or(tx.agent)], ['API', cut(routeKey(tx))], ['결과', tx.error ? '에러' : '정상'], ['상태', or(tx.status)],
    ['사용자', or(req.user)], ['IP', or(req.ip)], ['화면', req.referer ? pathOf(req.referer) : NONE],
  ];
  for (const s of Array.isArray(tx.spans) ? tx.spans : []) {
    if (!s || typeof s !== 'object') continue;
    if (s.type === 'mapper') list.push(['매퍼', cut(s.text)]);
    else if (s.type === 'http') list.push(['외부 호출', hostOf(s.text)]);
    else if (s.type === 'exception' && s.error) list.push(['예외', String(s.error).split(': ')[0].slice(0, MAX_EXCEPTION)]);
  }
  return new Map(list.map(([a, v]) => [`${a}\t${v}`, [a, v]]));
}

// 값마다 고른 쪽·나머지에서 그 값을 가진 요청 비율. 고른 쪽에 많은 순
function compare(picked, rest) {
  const out = { picked: picked.length, rest: rest.length, rows: [] };
  if (picked.length === 0 || rest.length === 0) return out;
  const counts = new Map();
  for (const [side, list] of [['picked', picked], ['rest', rest]]) {
    for (const tx of list) {
      for (const [key, [attr, value]] of attrs(tx)) {
        let c = counts.get(key);
        if (!c) counts.set(key, (c = { attr, value, picked: 0, rest: 0 }));
        c[side]++;
      }
    }
  }
  const rows = [];
  for (const c of counts.values()) {
    const diff = c.picked / picked.length - c.rest / rest.length;
    if (diff >= MIN_DIFF) rows.push({ c, diff });
  }
  rows.sort((a, b) => b.diff - a.diff || b.c.picked - a.c.picked);
  out.rows = rows.slice(0, MAX_ROWS).map(({ c }) => ({
    ...c, pickedPct: Math.round((100 * c.picked) / picked.length), restPct: Math.round((100 * c.rest) / rest.length),
  }));
  return out;
}

module.exports = { attrs, compare, routeKey };
```

```js
  // 본문을 MAX_BODY까지 읽어 JSON 객체면 then(body). 아니면 400
  function readJson(req, res, then) {
    const chunks = [];
    let size = 0;
    req.on('data', (c) => {
      size += c.length;
      if (size <= MAX_BODY) chunks.push(c); // 넘치면 읽기만 하고 버린다
    });
    req.on('end', () => {
      if (size > MAX_BODY) return send(res, 400, 'too large');
      let body;
      try {
        body = JSON.parse(Buffer.concat(chunks).toString('utf8'));
      } catch {
        return send(res, 400, 'bad json');
      }
      if (body === null || typeof body !== 'object') return send(res, 400, 'bad json');
      then(body);
    });
  }

  function ingest(req, res) {
    readJson(req, res, (body) => {
      pending.push(...store.ingest(body.agent, body.txs, body.active));
      send(res, 204);
    });
  }

  function compare(req, res) {
    readJson(req, res, (body) => {
      if (!Array.isArray(body.ids)) return send(res, 400, 'ids must be an array');
      const since = Number.isFinite(body.since) ? body.since : 0;
      json(res, 200, store.compare(body.ids, since, typeof body.agent === 'string' ? body.agent : ''));
    });
  }
```

```js
// ---- 무엇이 다른가 ----

function resetCompare() {
  compareSeq++;
  $('cmpinfo').textContent = '';
  $('compare').replaceChildren(COMPARE_HINT);
}

// 고른 점과 X-View에 보이는 나머지(같은 앱)를 수집 서버가 비교한다
async function fetchCompare(list) {
  const seq = ++compareSeq;
  $('compare').replaceChildren('비교하는 중…');
  let res;
  try {
    const r = await fetch('/compare', {
      method: 'POST', body: JSON.stringify({ ids: list.map((p) => p.id), since: Date.now() - WINDOW_MS, agent: selected }),
    });
    if (!r.ok) throw new Error(String(r.status));
    res = await r.json();
  } catch {
    if (seq === compareSeq) $('compare').replaceChildren('비교하지 못했습니다(수집 서버 연결 확인).');
    return;
  }
  if (seq === compareSeq) showCompare(res);
}

function showCompare(res) {
  $('cmpinfo').textContent = `고른 ${res.picked}건 vs 나머지 ${res.rest}건`;
  if (res.picked === 0 || res.rest === 0) return $('compare').replaceChildren('비교할 나머지 요청이 없습니다.');
  if (res.rows.length === 0) return $('compare').replaceChildren('뚜렷한 차이가 없습니다.');
  const table = document.createElement('table');
  table.append(row('th', ['속성', '값', '고른 쪽', '나머지']));
  for (const r of res.rows) {
    const tr = row('td', [r.attr, r.value]);
    tr.children[1].className = 'text';
    tr.append(pctCell(r.pickedPct, r.picked, ''), pctCell(r.restPct, r.rest, 'rest'));
    table.append(tr);
  }
  $('compare').replaceChildren(table);
}

// 0~100% 막대 + "85% (17건)"
function pctCell(pct, n, cls) {
  const td = document.createElement('td');
  const box = document.createElement('div');
  box.className = `pct ${cls}`;
  const track = document.createElement('span');
  track.className = 'track';
  const fill = document.createElement('i');
  fill.style.width = `${pct}%`;
  track.append(fill);
  box.append(track, `${pct}% (${n}건)`);
  td.append(box);
  return td;
}
```
