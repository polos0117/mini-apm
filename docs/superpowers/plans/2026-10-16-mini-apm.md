# 미니 APM 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** JENNIFER 같은 APM의 전체 흐름(에이전트 → 수집 서버 → X-View 대시보드)을 Node 22 기본 기능만으로 작게 만든다.

**Architecture:** 샘플 앱에 `--require`로 붙는 에이전트가 HTTP·SQL·fetch를 가로채 `AsyncLocalStorage`로 트랜잭션별 span을 모으고 1초마다 수집 서버에 POST한다. 수집 서버는 최근 10분을 메모리에 두고 SSE로 점·통계·처리 중 목록을 밀어 준다. 대시보드는 canvas 산점도와 상세 화면이다.

**Tech Stack:** Node 22.23 (`node:http`, `node:sqlite`, `node:async_hooks`, `node:test`), 브라우저 기본 API(canvas, EventSource). 외부 라이브러리 없음.

**Spec:** `docs/superpowers/specs/2026-10-16-mini-apm-design.md`

```js
// 수집 서버의 메모리 저장소. HTTP와 무관한 순수 로직만 둔다.
const KEEP_MS = 10 * 60 * 1000; // 트랜잭션 보관 기간
const STALE_MS = 5000;           // 이 시간 동안 소식 없는 에이전트는 처리 중 목록에서 뺀다

const end = (tx) => tx.start + tx.elapsed;
const isTx = (t) =>
  t !== null && typeof t === 'object' && typeof t.id === 'string' &&
  Number.isFinite(t.start) && Number.isFinite(t.elapsed) && Array.isArray(t.spans);
const toPoint = (tx) => ({ id: tx.id, name: tx.name, t: end(tx), elapsed: tx.elapsed, error: Boolean(tx.error) });

function p95(values) {
  if (values.length === 0) return 0;
  const sorted = [...values].sort((a, b) => a - b);
  return sorted[Math.ceil(0.95 * sorted.length) - 1];
}

function createStore() {
  // ponytail: 매초 전체를 훑는 O(n) 정리·통계. 10분치 수만 건까지는 충분, 넘치면 초 단위 버킷으로.
  let txs = [];
  const byId = new Map();
  const agents = new Map(); // 에이전트 이름 → { at, active }

  return {
    // 올바른 트랜잭션만 저장하고, 저장한 것의 점 목록을 돌려준다.
    ingest(agent, list, active, now = Date.now()) {
      const added = [];
      for (const tx of Array.isArray(list) ? list : []) {
        if (!isTx(tx) || byId.has(tx.id)) continue;
        txs.push(tx);
        byId.set(tx.id, tx);
        added.push(toPoint(tx));
      }
      agents.set(String(agent), { at: now, active: Array.isArray(active) ? active : [] });
      return added;
    },
    get(id) {
      return byId.get(id) ?? null;
    },
    points(since) {
      return txs.filter((tx) => end(tx) >= since).map(toPoint);
    },
    stats(secStart) {
      const inSec = txs.filter((tx) => end(tx) >= secStart && end(tx) < secStart + 1000);
      const elapsed = inSec.map((tx) => tx.elapsed);
      return {
        tps: inSec.length,
        avg: elapsed.length ? Math.round(elapsed.reduce((a, b) => a + b, 0) / elapsed.length) : 0,
        p95: Math.round(p95(elapsed)),
        errors: inSec.filter((tx) => tx.error).length,
      };
    },
    active(now = Date.now()) {
      const all = [];
      for (const a of agents.values()) if (now - a.at <= STALE_MS) all.push(...a.active);
      return all.sort((x, y) => y.elapsed - x.elapsed);
    },
    prune(now = Date.now()) {
      const cutoff = now - KEEP_MS;
      const keep = [];
      for (const tx of txs) {
        if (end(tx) >= cutoff) keep.push(tx);
        else byId.delete(tx.id);
      }
      txs = keep;
    },
  };
}

module.exports = { createStore, p95 };
```

```js
// 대시보드 계산 함수. 브라우저(app.mjs)와 테스트가 함께 쓴다.

// rect = { t0, t1: 끝난 시각 범위, e0, e1: 응답시간 범위 }. 끄는 방향은 상관없다.
export function pickInRect(points, rect) {
  const t0 = Math.min(rect.t0, rect.t1), t1 = Math.max(rect.t0, rect.t1);
  const e0 = Math.min(rect.e0, rect.e1), e1 = Math.max(rect.e0, rect.e1);
  return points
    .filter((p) => p.t >= t0 && p.t <= t1 && p.elapsed >= e0 && p.elapsed <= e1)
    .sort((a, b) => b.elapsed - a.elapsed);
}

const STEPS = [1000, 3000, 5000, 10000];

// 세로축 최댓값(ms)
export function yMax(maxElapsed) {
  return STEPS.find((s) => s >= maxElapsed) ?? Math.ceil(maxElapsed / 1000) * 1000;
}
```

```js
const fs = require('node:fs');
const path = require('node:path');
const { createStore } = require('./store');

const DASH = path.join(__dirname, '..', 'dashboard');
const TYPES = {
  '.html': 'text/html; charset=utf-8',
  '.mjs': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
};
const MAX_BODY = 1024 * 1024;
const BACKLOG_MS = 5 * 60 * 1000; // 새로 연결한 화면에 보내는 과거 점
const STATS_LAG_MS = 2000;        // 에이전트가 1초마다 묶어 보내므로 2초 전의 1초를 '마감된 1초'로 본다

function send(res, status, body = '', type = 'text/plain; charset=utf-8') {
  if (res.headersSent) return;
  res.writeHead(status, { 'Content-Type': type });
  res.end(body);
}
const json = (res, status, obj) => send(res, status, JSON.stringify(obj), 'application/json; charset=utf-8');
const sse = (msg) => `data: ${JSON.stringify(msg)}\n\n`;
const statSec = (now) => Math.floor((now - STATS_LAG_MS) / 1000) * 1000;

function createCollector(store = createStore()) {
  const clients = new Set();
  let pending = []; // 다음 틱에 보낼 새 점

  function ingest(req, res) {
    const chunks = [];
    let size = 0;
    req.on('data', (c) => {
      size += c.length;
      if (size <= MAX_BODY) chunks.push(c); // 넘치면 읽기만 하고 버린다
    });
    req.on('end', () => {
      if (size > MAX_BODY) return send(res, 400, 'too large');
      let body;
      try {
        body = JSON.parse(Buffer.concat(chunks).toString('utf8'));
      } catch {
        return send(res, 400, 'bad json');
      }
      if (body === null || typeof body !== 'object') return send(res, 400, 'bad json');
      pending.push(...store.ingest(body.agent, body.txs, body.active));
      send(res, 204);
    });
  }

  function stream(req, res) {
    res.writeHead(200, { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', Connection: 'keep-alive' });
    const now = Date.now();
    res.write(sse({ points: store.points(now - BACKLOG_MS), stats: store.stats(statSec(now)), active: store.active(now) }));
    clients.add(res);
    res.on('close', () => clients.delete(res)); // req의 close는 요청 본문이 끝나면 바로 오므로 res로 본다
  }

  function serveFile(pathname, res) {
    const file = path.join(DASH, pathname === '/' ? 'index.html' : pathname);
    const type = TYPES[path.extname(file)];
    if (!file.startsWith(DASH + path.sep) || !type) return send(res, 404, 'not found');
    fs.readFile(file, (err, data) => (err ? send(res, 404, 'not found') : send(res, 200, data, type)));
  }

  const server = http.createServer((req, res) => {
    const { pathname } = new URL(req.url, 'http://x');
    if (req.method === 'POST' && pathname === '/ingest') return ingest(req, res);
    if (req.method !== 'GET') return send(res, 404, 'not found');
    if (pathname === '/stream') return stream(req, res);
    if (pathname.startsWith('/tx/')) {
      const tx = store.get(pathname.slice(4));
      return tx ? json(res, 200, tx) : send(res, 404, 'not found');
    }
    serveFile(pathname, res);
  });

  const timer = setInterval(() => {
    const now = Date.now();
    store.prune(now);
    const msg = sse({ points: pending, stats: store.stats(statSec(now)), active: store.active(now) });
    pending = [];
    for (const res of clients) res.write(msg);
  }, 1000);
  timer.unref();

  return {
    server,
    close() {
      clearInterval(timer);
      for (const res of clients) res.end();
      server.closeAllConnections();
      server.close();
    },
  };
}

if (require.main === module) {
  const port = Number(process.env.PORT) || 7700;
  createCollector().server.listen(port, () => console.log(`수집 서버: http://localhost:${port}`));
}

module.exports = { createCollector };
```

```js
// 미니 APM 에이전트. `node --require ./agent app.js`로 앱보다 먼저 로드된다.
// JENNIFER가 -javaagent로 바이트코드를 바꾸듯, 여기서는 Node 내장 함수를 감싸서 계측한다.
const http = require('node:http');
const { DatabaseSync } = require('node:sqlite');
const { AsyncLocalStorage } = require('node:async_hooks');
const { randomUUID } = require('node:crypto');

const COLLECTOR = process.env.APM_COLLECTOR || 'http://localhost:7700';
const AGENT = process.env.APM_AGENT || 'sample';
const INTERVAL = Number(process.env.APM_INTERVAL) || 1000;
const MAX_BUFFER = 5000;

const als = new AsyncLocalStorage(); // 비동기 흐름을 따라 '지금 어느 트랜잭션인가'를 들고 다닌다
const open = new Map();              // 처리 중인 트랜잭션
let done = [];                       // 전송 대기
const rawFetch = globalThis.fetch;   // 전송용: 자기 자신은 계측하지 않는다
const round = (ms) => Math.round(ms * 10) / 10;

// 에이전트 자신의 오류가 앱으로 번지지 않게 한다.
function safe(fn) {
  try {
    fn();
  } catch (err) {
    console.error('[apm]', err.message);
  }
}

// ---- 트랜잭션 ----

function begin(req) {
  const tx = {
    id: randomUUID(),
    name: `${req.method} ${req.url.split('?')[0]}`,
    start: Date.now(),
    t0: performance.now(),
    status: 0,
    error: null,
    spans: [],
  };
  open.set(tx.id, tx);
  return tx;
}

function finish(tx, status) {
  if (!open.delete(tx.id)) return; // finish와 close 중 먼저 온 쪽만
  const { t0, ...rec } = tx;
  rec.elapsed = round(performance.now() - t0);
  rec.status = status;
  if (status >= 500) rec.error = tx.spans.findLast((s) => s.error)?.error ?? `HTTP ${status}`;
  done.push(rec);
  if (done.length > MAX_BUFFER) done.splice(0, done.length - MAX_BUFFER);
}

const origEmit = http.Server.prototype.emit;
http.Server.prototype.emit = function (event, req, res) {
  if (event !== 'request') return origEmit.apply(this, arguments);
  let tx = null;
  safe(() => {
    tx = begin(req);
    const end = () => safe(() => finish(tx, res.statusCode));
    res.once('finish', end);
    res.once('close', end);
  });
  return tx ? als.run(tx, () => origEmit.apply(this, arguments)) : origEmit.apply(this, arguments);
};

// ---- span: 트랜잭션 안에서 일어난 SQL·외부 호출 ----

// fn을 실행하며 시간을 잰다. 반환값·예외는 그대로 돌려준다.
// failOf(결과)는 정상 반환이지만 실패로 볼 결과의 메시지를 돌려준다(fetch의 500 응답).
function span(type, text, fn, failOf = () => null) {
  const tx = als.getStore();
  if (!tx || !open.has(tx.id)) return fn();
  const s = { type, text: String(text), offset: round(performance.now() - tx.t0), elapsed: 0, error: null };
  tx.spans.push(s);
  const t = performance.now();
  const stop = (err) => {
    s.elapsed = round(performance.now() - t);
    if (err) s.error = err;
  };
  let out;
  try {
    out = fn();
  } catch (err) {
    stop(err.message);
    throw err;
  }
  if (out instanceof Promise) {
    return out.then(
      (v) => (stop(failOf(v)), v),
      (err) => {
        stop(err.message);
        throw err;
      },
    );
  }
  stop(failOf(out));
  return out;
}

// SQL: 문장 객체(StatementSync)의 프로토타입은 처음 prepare한 결과에서 얻는다.
const origPrepare = DatabaseSync.prototype.prepare;
let statementPatched = false;
DatabaseSync.prototype.prepare = function () {
  const stmt = origPrepare.apply(this, arguments);
  if (!statementPatched) {
    statementPatched = true;
    safe(() => patchStatement(Object.getPrototypeOf(stmt)));
  }
  return stmt;
};

function patchStatement(proto) {
  for (const name of ['run', 'get', 'all']) {
    const orig = proto[name];
    proto[name] = function () {
      return span('sql', this.sourceSQL, () => orig.apply(this, arguments));
    };
  }
}

const origExec = DatabaseSync.prototype.exec;
DatabaseSync.prototype.exec = function (sql) {
  return span('sql', sql, () => origExec.apply(this, arguments));
};

// 나가는 호출
globalThis.fetch = function (input, init) {
  const method = (init?.method || input?.method || 'GET').toUpperCase();
  const url = input?.url ?? String(input);
  return span('http', `${method} ${url}`, () => rawFetch(input, init), (r) => (r.status >= 500 ? `HTTP ${r.status}` : null));
};

// ---- 전송 ----

function activeList() {
  const now = performance.now();
  return [...open.values()].map((tx) => ({
    id: tx.id,
    name: tx.name,
    elapsed: round(now - tx.t0),
    lastSpan: tx.spans.at(-1)?.text ?? null,
  }));
}

function flush() {
  safe(() => {
    const txs = done;
    done = [];
    rawFetch(`${COLLECTOR}/ingest`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ agent: AGENT, txs, active: activeList() }),
    }).catch(() => {}); // 수집 서버가 없으면 이 묶음은 버린다
  });
}

setInterval(flush, INTERVAL).unref();
```

```js
function draw() {
  const w = canvas.clientWidth, h = canvas.clientHeight, dpr = devicePixelRatio || 1;
  if (canvas.width !== w * dpr || canvas.height !== h * dpr) {
    canvas.width = w * dpr;
    canvas.height = h * dpr;
  }
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  ctx.clearRect(0, 0, w, h);

  const now = Date.now();

  const top = yMax(max);

    x: (t) => PAD.l + ((t - (now - WINDOW_MS)) / WINDOW_MS) * pw,
    y: (ms) => PAD.t + ph - (ms / top) * ph,
    t: (x) => now - WINDOW_MS + ((x - PAD.l) / pw) * WINDOW_MS,
    ms: (y) => ((PAD.t + ph - y) / ph) * top,
  };

  ctx.font = '11px system-ui';
  ctx.strokeStyle = COLOR.grid;
  ctx.fillStyle = COLOR.dim;

  ctx.textBaseline = 'middle';

    const ms = (top / 4) * i, y = scale.y(ms);
    ctx.beginPath();
    ctx.moveTo(PAD.l, y);
    ctx.lineTo(w - PAD.r, y);
    ctx.stroke();
    ctx.fillText(String(Math.round(ms)), PAD.l - 6, y);
  }
  ctx.textAlign = 'center';
  ctx.textBaseline = 'top';

async function showDetail(id) {
  const box = $('detail');
  const res = await fetch(`/tx/${encodeURIComponent(id)}`);
  if (!res.ok) return (box.textContent = '이미 정리된 트랜잭션입니다(10분 경과).');

  const self = Math.max(0, tx.elapsed - tx.spans.reduce((sum, s) => sum + s.elapsed, 0));

  const head = document.createElement('p');
  head.textContent = `${tx.name}   ${Math.round(tx.elapsed)}ms   상태 ${tx.status}${tx.error ? `   — ${tx.error}` : ''}`;
  if (tx.error) head.className = 'err';

  const table = document.createElement('table');
  for (const s of tx.spans) {

    table.append(spanRow(s.type === 'sql' ? 'SQL' : '외부', label, s.offset / total, s.elapsed, s.elapsed / total, s.error));
  }
  table.append(spanRow('앱', '앱 자체 처리', null, self, 0, null));

}

// left·width는 전체 대비 비율(0~1). left가 null이면 막대 없이 시간만.
function spanRow(kind, text, left, elapsed, width, error) {
  const tr = row('td', [kind, text]);
  tr.children[1].className = 'text';
  if (error) tr.className = 'err';
  const barCell = document.createElement('td');
  const bar = document.createElement('div');
  bar.className = 'bar';

    fill.style.width = `${Math.max(0.5, width * 100)}%`;
    if (error) fill.className = 'err';
    bar.append(fill);
  }
  barCell.append(bar);
  const ms = document.createElement('td');
  ms.className = 'num';
  ms.textContent = `${Math.round(elapsed)}ms`;
  tr.append(barCell, ms);
  return tr;
}
```

````markdown
| 에이전트의 `try/catch`, 버퍼 상한 | 에이전트 안정성 | 모니터링 도구가 앱을 망가뜨리면 안 된다 |

## 샘플 앱의 일부러 넣은 증상

| 경로 | 증상 | 상세 화면에서 보이는 모습 |
|---|---|---|
| `GET /orders` | 상관 하위 쿼리 + 인덱스 없음 | SQL 막대가 대부분 |
| `POST /checkout` | 느리고 가끔 실패하는 결제 호출 | 외부 막대가 대부분, 실패 시 빨강 |
| `GET /report` | 무거운 계산 | "앱 자체 처리"가 대부분 |
| `GET /boom` | 예외 → 500 | 에러 점 |

`node:sqlite`는 동기식이라 느린 SQL 동안 다른 요청도 멈춘다. X-View에서 점들이 같은 시각에 한꺼번에 쌓이는 모습으로 보인다.
````
